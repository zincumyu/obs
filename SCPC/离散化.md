#include <bits/stdc++.h>
using namespace std;

// 将原数组 a 离散化，结果存到 rank 中
// rank[i] 表示 a[i] 在所有不同数值中的排名（1 ~ m）
vector<int> discretize(vector<int> a) {
    vector<int> tmp = a;
    sort(tmp.begin(), tmp.end());
    tmp.erase(unique(tmp.begin(), tmp.end()), tmp.end());
    
    vector<int> rank(a.size());
    for (int i = 0; i < (int)a.size(); i++) {
        rank[i] = lower_bound(tmp.begin(), tmp.end(), a[i]) - tmp.begin() + 1;
    }
    return rank;
}

int main() {
    vector<int> a = {100, -5, 3, 3, 1000000000};
    vector<int> rk = discretize(a);
    for (int x : rk) cout << x << " ";
    // 输出：4 1 2 2 3
    return 0;
}
