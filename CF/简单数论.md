
## 1.质素判断
1.判偶数 后循环奇数 i * i<=x

```cpp
bool isprime(ll x){

    if(x<=1) return false;

    if(x==2) return true;

    if(x%2==0) return false;

    for(ll i=3;i*i<=x;i+=2){

        if(x%i==0) return false;

    }

    return true;

}
```


2.遍历求质数
```cpp
const int MAX = 1000005;
vector<bool> pri(MAX, true);

void sieve()
{
    pri[0]=pri[1]=false;
    for(int i=2;i<MAX;i++)
    {
        if(pri[i])
        {
            for(int j=i+i;j<MAX;j+=i)
                pri[j]=false;
        }
    }
}
```
